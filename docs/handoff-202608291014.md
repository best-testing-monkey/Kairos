SELECT strategy_name, COUNT(DISTINCT run_id) as n_groups, SUM(signal_count) as total_signals,

FROM model_results WHERE stage='base' AND signal_count >= 3
GROUP BY strategy_name ORDER BY w_sharpe DESC
